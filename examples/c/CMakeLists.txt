# Build and run the C and C++ examples against the wickra-genome C ABI.
#
#   cargo build --release -p wickra-genome-c
#   cmake -S examples/c -B examples/c/build
#   cmake --build examples/c/build --config Release
#   ctest --test-dir examples/c/build -C Release --output-on-failure
#
# WICKRA_GENOME_LIB_DIR defaults to the workspace target/release directory and can
# be overridden for out-of-tree builds.

cmake_minimum_required(VERSION 3.15)
project(wickra_genome_examples C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(WICKRA_GENOME_LIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../target/release"
    CACHE PATH "Directory containing the built wickra-genome C ABI library")

set(WICKRA_GENOME_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../bindings/c/include")

find_library(WICKRA_GENOME_LIB
    NAMES wickra_genome libwickra_genome
    HINTS "${WICKRA_GENOME_LIB_DIR}"
    REQUIRED)

add_executable(genome genome.c)
target_include_directories(genome PRIVATE "${WICKRA_GENOME_INCLUDE}")
target_link_libraries(genome PRIVATE "${WICKRA_GENOME_LIB}")

add_executable(genomepp genome.cpp)
target_include_directories(genomepp PRIVATE "${WICKRA_GENOME_INCLUDE}")
target_link_libraries(genomepp PRIVATE "${WICKRA_GENOME_LIB}")

add_executable(streaming_test streaming_test.c)
target_include_directories(streaming_test PRIVATE "${WICKRA_GENOME_INCLUDE}")
target_link_libraries(streaming_test PRIVATE "${WICKRA_GENOME_LIB}")

# C has no directory API portable between POSIX and Windows, so the spec list is
# globbed here at configure time and written into golden_specs.h. That keeps the
# property the other bindings get from a runtime glob: a spec added to the corpus
# is covered without editing golden_test.c.
set(GOLDEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../golden")
file(GLOB GOLDEN_SPEC_FILES CONFIGURE_DEPENDS "${GOLDEN_DIR}/specs/*.json")
set(GOLDEN_SPEC_ENTRIES "")
foreach(spec_file ${GOLDEN_SPEC_FILES})
    get_filename_component(spec_name "${spec_file}" NAME)
    string(APPEND GOLDEN_SPEC_ENTRIES "    \"${spec_name}\",
")
endforeach()
list(LENGTH GOLDEN_SPEC_FILES GOLDEN_SPEC_COUNT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/golden_specs.h"
"/* Generated by examples/c/CMakeLists.txt -- do not edit. */
"
"#ifndef WICKRA_GENOME_GOLDEN_SPECS_H
"
"#define WICKRA_GENOME_GOLDEN_SPECS_H
"
"#include <stddef.h>
"
"#define GOLDEN_DIR \"${GOLDEN_DIR}\"
"
"static const char *GOLDEN_SPECS[] = {
${GOLDEN_SPEC_ENTRIES}};
"
"static const size_t GOLDEN_SPEC_COUNT = ${GOLDEN_SPEC_COUNT};
"
"#endif
")

add_executable(golden_test golden_test.c)
target_include_directories(golden_test PRIVATE
    "${WICKRA_GENOME_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(golden_test PRIVATE "${WICKRA_GENOME_LIB}")

# On Windows the runtime DLL must sit next to the executable (there is no rpath);
# without it the loader pops a modal "DLL not found" dialog that hangs ctest. Copy
# it next to each example after build. (Unix resolves the .so/.dylib via the
# linker's embedded path, so this is Windows-only.)
if(WIN32)
    foreach(example genome genomepp streaming_test golden_test)
        add_custom_command(TARGET ${example} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${WICKRA_GENOME_LIB_DIR}/wickra_genome.dll"
                "$<TARGET_FILE_DIR:${example}>"
            VERBATIM)
    endforeach()
endif()

enable_testing()
add_test(NAME genome COMMAND genome)
add_test(NAME genomepp COMMAND genomepp)
add_test(NAME streaming_test COMMAND streaming_test)
add_test(NAME golden_test COMMAND golden_test)
# A missing runtime dependency must fail fast, never block the runner for hours.
set_tests_properties(genome genomepp streaming_test golden_test PROPERTIES TIMEOUT 60)
